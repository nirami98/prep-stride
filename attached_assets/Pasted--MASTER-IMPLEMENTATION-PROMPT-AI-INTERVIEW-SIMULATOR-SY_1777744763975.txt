🔷 DATA SCHEMA (PRISMA)
model User {
  id        String   @id @default(uuid())
  email     String   @unique
  password  String
  createdAt DateTime @default(now())
  sessions  InterviewSession[]
}

model InterviewSession {
  id        String   @id @default(uuid())
  userId    String
  user      User     @relation(fields: [userId], references: [id])
  startedAt DateTime @default(now())
  endedAt   DateTime?
  answers   Answer[]
  feedback  Feedback?
}

model Question {
  id       String @id @default(uuid())
  content  String
  category String
}

model Answer {
  id          String   @id @default(uuid())
  sessionId   String
  questionId  String
  response    String
  createdAt   DateTime @default(now())
}

model Feedback {
  id          String   @id @default(uuid())
  sessionId   String   @unique
  score       Float
  strengths   String
  weaknesses  String
  suggestions String
}

model Analytics {
  id        String   @id @default(uuid())
  userId    String
  avgScore  Float
  totalSessions Int
  updatedAt DateTime @updatedAt
}
